cmake_minimum_required(VERSION 3.16)
project(DualPackageConsumer LANGUAGES C)
enable_testing()

function(load_package package version exact target)
  # Search only the Config directory produced by this case's installation.
  set(config_dir "${D21_PREFIX}/lib/cmake/${package}")
  if(exact)
    set(exact_flag EXACT)
  endif()
  find_package(${package} ${version} ${exact_flag} CONFIG REQUIRED
    PATHS "${config_dir}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  if(NOT "${${package}_DIR}" STREQUAL "${config_dir}")
    message(FATAL_ERROR "Unexpected ${package} config: ${${package}_DIR}")
  endif()
  if(package STREQUAL "ProbSpan")
    set(expected_version "1.1.0")
    set(archive_name probspan)
  else()
    set(expected_version "2.0.0")
    set(archive_name probspan_models)
  endif()
  if(NOT "${${package}_VERSION}" STREQUAL "${expected_version}")
    message(FATAL_ERROR "Unexpected ${package} version")
  endif()
  get_target_property(archive "${target}" IMPORTED_LOCATION_RELEASE)
  if(NOT archive STREQUAL "${D21_PREFIX}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}${archive_name}${CMAKE_STATIC_LIBRARY_SUFFIX}"
      OR NOT EXISTS "${archive}")
    message(FATAL_ERROR "Unexpected imported archive: ${archive}")
  endif()
  if(package STREQUAL "ProbSpanModels")
    get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
    set(expected_dependencies "m")
    if(MSVC)
      set(expected_dependencies "dependencies-NOTFOUND")
    endif()
    if(NOT dependencies STREQUAL expected_dependencies)
      message(FATAL_ERROR "Unexpected Models dependencies: ${dependencies}")
    endif()
  endif()
endfunction()

if(USE_MODELS AND MODELS_FIRST)
